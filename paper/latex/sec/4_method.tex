% 고정 iteration / 고정 FPS / maturity gate 가 증가를 pool 크기·하드웨어에 종속시킨다.
\slot{P1 \textemdash\ DIAGNOSIS (taming3dgs \S3.2)}{12}{%
  A fixed per-keyframe iteration count, a fixed frame rate, and a maturity gate all tie the
  growth of the training view set to the size of that set and to hardware speed. The result is
  update dilution that cannot be predicted in advance.}

% 왜 wall-clock 도 iteration 도 아니고 "완료된 service" 인가를 여기서 답한다.
\slot{P2 \textemdash\ MEASUREMENT + LINEAGE (taming3dgs; RFC 2212)}{14}{%
  Introduce completed GPU service $S(t)$, and answer here why it is neither wall-clock time nor
  an iteration count. Lineage sentence: a token bucket whose tokens are minted by \emph{completed
  GPU service} rather than by wall-clock time, which makes it credit-based rather than
  rate-based. Point at Fig.~\ref{fig:growthtrace}.}

\slot{P3 \textemdash\ MECHANISM}{5}{%
  Introduce (\ref{eq:astar}) and (\ref{eq:bucket}); say what each symbol is and why the bucket
  is constrained to stay non-negative.}
\begin{align}
  A^{*}(t) &= \min\!\Big(M(t),\; A_0 + \Big\lfloor \tfrac{B_0+\gamma S(t)}{\kappa} \Big\rfloor \Big) \label{eq:astar}\\
  B_t &= B_0 + \gamma S(t) - \kappa\big(|A_t| - A_0\big) \;\ge\; 0 \label{eq:bucket}
\end{align}

% carry vs no-prepurchase 를 여기서 두 문장으로 종결한다. taming3dgs §3.2 4문단 형식.
\slot{P4 \textemdash\ CORRECTION RULE + FORWARD POINTER}{10}{%
  Realized growth after prunes and removals, (\ref{eq:quota}).
  \textbf{\color{red}Close carry vs.\ no-prepurchase here, in two sentences.}
  End with: \S\ref{sec:exp} shows the law holds with zero integer error, and how quality
  degrades gracefully as $\kappa$ is lowered.}
\begin{equation}\label{eq:quota}
  Q_t = \min\!\big(|C_t|,\; [\,A^{*}(t) - |A_{t^-}|\,]_+\big)
\end{equation}

\subsection{Entropy-Regularized Count Balancing}
\label{ssec:ercb}

% 소절 개수와 유도 단계 수를 일치시킨다.
\slot{P1 \textemdash\ ROADMAP, enumerated (chen2026cover \S4)}{9}{%
  The derivation proceeds in three steps: (1) express selection-count imbalance as a potential
  $\Phi$; (2) solve an entropy-regularized Lagrangian for the unique distribution; (3) extend it
  to a $K$-view block drawn without replacement.}

\slot{P2 \textemdash\ FORMULATING}{8}{%
  Equation (\ref{eq:phi}). State that the last term is common to all $i$ and therefore does not
  affect the $\arg\max$.}
\begin{equation}\label{eq:phi}
  \Phi(\mathbf{n}) = \tfrac12\sum_i (n_i-\bar n)^2,
  \qquad
  \Delta_i\Phi = n_i - \bar n + \tfrac12\big(1-\tfrac1N\big)
\end{equation}

\slot{P3 \textemdash\ DERIVING}{10}{%
  Entropy-regularized Lagrangian to (\ref{eq:gibbs}). \textbf{Say ``unique solution'' explicitly.}
  Explain the $\beta=1/\tau$ reparameterization.}
\begin{equation}\label{eq:gibbs}
  p_t(i) = \frac{\exp(-\beta n_i)}{\sum_{j}\exp(-\beta n_j)},\qquad \beta = 1/\tau
\end{equation}

% 근사마다 성립 조건을 붙인다 — chen2026cover §4.4 의 가정 원장 형식.
\slot{P4 \textemdash\ EXTENSION + ASSUMPTION LEDGER}{13}{%
  Extend to a $K$-view sequential weighted draw without replacement, holding the count snapshot
  $n_i^{(b)}$ fixed within a block. \textbf{State that this is a first-order approximation inside
  the block, with the condition under which it holds.} Measure the retained conditional entropy
  with $\rho_H$ (\ref{eq:rhoH}).}
\begin{equation}\label{eq:rhoH}
  \rho_H = \frac{\sum_{b,k} H(p_{b,k})}{\sum_{b,k}\log|W_b^{(k)}|}
\end{equation}

% 네 가지가 반드시 들어간다. (d) 가 in-method 실패 보고 — lmrs §4.2 형식.
\slot{P5 \textemdash\ SCOPE GUARD + LINEAGE + FAILURE}{20}{%
  (a) Equality-to-bound audit: with $\beta{=}0$ this coincides with random reshuffling
  \emph{only} when $K \ge N_t$; we do not inherit its convergence guarantees, since the set grows
  and the objective is non-convex.
  (b) Name the draw: sequential weighted sampling without replacement, i.e.\ Plackett--Luce.
  (c) Name the precedent: count-based priorities appear in experience replay; our objective is
  the opposite of loss-prioritized data selection.
  (d) \textbf{\color{red}In-method failure}: this equalizes selection counts but not lifetime
  exposure (rot mid/first $0.758 \to 0.520$). Using counts alone as the potential is a design
  choice and its limit.}
