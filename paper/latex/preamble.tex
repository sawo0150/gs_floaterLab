\newcommand{\slot}[3]{%

  \begingroup
    \setlength{\fboxsep}{4pt}\setlength{\fboxrule}{0.4pt}%
    \fcolorbox{slotgray}{slotbg}{%
      \parbox[t][#2\baselineskip][t]{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
        {\sffamily\scriptsize\bfseries\color{skelblue}#1}%
        \hfill{\sffamily\scriptsize\color{slotgray}#2 lines}\\[1pt]
        {\sffamily\scriptsize\color{slotgray}#3\par}%
      }}%
  \endgroup
  \par\vspace{3pt}%
}
